cmake_minimum_required(VERSION 3.20)
project(dstmc_native LANGUAGES C CXX)
if(POLICY CMP0135)
 cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
add_executable(publisher_rgba_reader ../mc/src/contractTest/cpp/publisher_rgba_reader.cpp)
target_include_directories(publisher_rgba_reader PRIVATE include)
if(APPLE)
 set_target_properties(publisher_rgba_reader PROPERTIES OSX_ARCHITECTURES x86_64)
endif()
add_library(dstmc_contracts STATIC src/camera_service.cpp)
target_include_directories(dstmc_contracts PUBLIC include windows)
add_executable(windows_contract_test windows/tests/contracts.cpp)
target_link_libraries(windows_contract_test PRIVATE dstmc_contracts)
add_test(NAME windows_contract COMMAND windows_contract_test)
set_tests_properties(windows_contract PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
if(MSVC)
 target_compile_options(windows_contract_test PRIVATE /UNDEBUG)
else()
 target_compile_options(windows_contract_test PRIVATE -UNDEBUG)
endif()
if(WIN32)
 target_link_libraries(publisher_rgba_reader PRIVATE shell32)
 if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows backend requires x64")
 endif()
 if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
  set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
 endif()
 include(FetchContent)
 FetchContent_Declare(minhook URL https://codeload.github.com/TsudaKageyu/minhook/tar.gz/c3fcafdc10146beb5919319d0683e44e3c30d537 URL_HASH SHA256=31757833475be039c16359d4e276e718db6ba2ba61a6b870053a112106015822)
 FetchContent_MakeAvailable(minhook)
 # JSON is header only; pin its tag in the same way as MinHook.
 FetchContent_Declare(json_headers URL https://codeload.github.com/nlohmann/json/tar.gz/9cca280a4d0ccf0c08f47a99aa71d1b0e52f8d03 URL_HASH SHA256=0dbc5e40a01ff142e7e68c03e85247a4dcede2f592d12d3677dee3664d17975a)
 FetchContent_MakeAvailable(json_headers)
 add_library(loader_fixture SHARED windows/tests/loader_fixture.cpp)
 set_target_properties(loader_fixture PROPERTIES PREFIX "" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_executable(windows_loader_http_smoke windows/tests/loader_http_smoke.cpp)
 target_link_libraries(windows_loader_http_smoke PRIVATE ws2_32 shell32)
 target_compile_definitions(windows_loader_http_smoke PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 add_test(NAME windows_loader_http_smoke COMMAND windows_loader_http_smoke $<TARGET_FILE:dstmc_bridge>)
 set_tests_properties(windows_loader_http_smoke PROPERTIES TIMEOUT 30 RESOURCE_LOCK dstmc_loopback)
 add_library(dstmc_bridge SHARED windows/backend.cpp windows/render.cpp)
 target_include_directories(dstmc_bridge PRIVATE windows/third_party)
 target_link_libraries(dstmc_bridge PRIVATE dstmc_contracts minhook nlohmann_json::nlohmann_json ws2_32 winhttp bcrypt user32)
 target_compile_definitions(dstmc_bridge PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
 set_target_properties(dstmc_bridge PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
 foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(dstmc_bridge PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}")
 endforeach()
 if(MINGW)
  foreach(target dstmc_bridge publisher_rgba_reader windows_contract_test windows_loader_http_smoke loader_fixture)
   target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++ -static)
  endforeach()
 endif()
endif()
